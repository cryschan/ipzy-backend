// RecommendationService - AI 추천 요청 및 저장
public List<Recommendation> requestAndSaveRecommendations(Long sessionId) {
    QuizSession session = quizSessionRepository.findByIdWithAnswers(sessionId)
            .orElseThrow(() -> new QuizException(QuizErrorCode.SESSION_NOT_FOUND));

    AiRecommendationRequest request = buildAiRequest(session);
    AiRecommendationResponse response = pythonAiClient.requestRecommendation(request);

    return convertAndSave(session, response);
}
